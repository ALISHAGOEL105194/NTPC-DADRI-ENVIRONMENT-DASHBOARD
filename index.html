// ============================================= // NTPC DADRI CEMS — Google Apps Script // Supports both JSON and JSONP (no CORS issues) // ============================================= var FILE_ID = 'YOUR_FILE_ID'; // ← paste your Google Drive file ID here function doGet(e) { var callback = e.parameter.callback || ''; try { var file = DriveApp.getFileById(FILE_ID); var blob = file.getBlob(); var b64 = Utilities.base64Encode(blob.getBytes()); var result = JSON.stringify({ status : 'ok', data : b64, filename : file.getName(), updated : new Date().toISOString() }); // Return JSONP if callback param present (used by dashboard) if (callback) { return ContentService .createTextOutput(callback + '(' + result + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } // Return plain JSON otherwise return ContentService .createTextOutput(result) .setMimeType(ContentService.MimeType.JSON); } catch (err) { var errResult = JSON.stringify({ status : 'error', error : err.message }); if (callback) { return ContentService .createTextOutput(callback + '(' + errResult + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(errResult) .setMimeType(ContentService.MimeType.JSON); } } // ── HOW TO DEPLOY (do this once) ─────────────── // 1.
